          <Video
            ref={videoRef}
            source={{ 
              uri: sourceUri,
              // 🔧 CRITICAL: Android SAF compatibility flags for ExoPlayer

                console.log(`[Viewer] ✅ Video loaded: ${currentItem.id}`);
                setIsReady(true);
                Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
              }
            }}
            onBuffer={({ isBuffering }) => {
              if (mountedRef.current && isBuffering && !isReady) {
                // Optional: show buffering indicator

### ✅ Step 2: Fix `prepareStatusForViewing` in `MediaContext.tsx`

Ensure it returns URI immediately for playback (non-blocking):

```typescript
// contexts/MediaContext.tsx — UPDATE prepareStatusForViewing

const prepareStatusForViewing = useCallback(async (
  item: StatusItem,
  opts?: { forShare?: boolean; forPlayback?: boolean },
): Promise<string> => {
  // Local files: return immediately
  if (!item.uri.startsWith('content://')) return item.uri;

  // 🔧 CRITICAL: For playback, return URI immediately + start background copy
  if (!opts?.forShare) {
    if (opts?.forPlayback) {
      // Start background copy silently (for fallback if direct playback fails)
      enqueueCopy(async () => {
        const ext = item.name.split('.').pop() || 'mp4';
        const safeId = item.id.replace(/[:\/\\?%*|"<>]/g, '_');
        const cacheUri = `${FileSystem.cacheDirectory}playback_${safeId}.${ext}`;
        
        const info = await FileSystem.getInfoAsync(cacheUri).catch(() => ({ exists: false }));
        if (!info.exists) {
          await FileSystem.copyAsync({ from: item.uri, to: cacheUri }).catch(() => {});
        }
      });
      
      logDirectPlaySuccess();
      return item.uri; // Return immediately → no navigation delay
    }
    return item.uri;
  }

  // Share path: still await copy (share sheet needs real file)
  const ext = item.name.split('.').pop() || 'jpg';
  const safeId = item.id.replace(/[:\/\\?%*|"<>]/g, '_');
  const tempUri = `${FileSystem.cacheDirectory}share_${safeId}.${ext}`;
  
  try {
    const info = await FileSystem.getInfoAsync(tempUri);
    if (info.exists && (info as any).size > 0) return tempUri;
  } catch {}

  return enqueueCopy(async () => {
    await FileSystem.copyAsync({ from: item.uri, to: tempUri });
    return tempUri;
  });
}, []);
```
