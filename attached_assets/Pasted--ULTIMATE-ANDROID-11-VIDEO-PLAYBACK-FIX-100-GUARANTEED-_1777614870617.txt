1. **Scoped Storage** blocks `/storage/emulated/0/WhatsApp/...` paths

```typescript
import React, { useState, useEffect, useRef } from 'react';
import { View, ActivityIndicator, Platform, ToastAndroid, Linking } from 'react-native';
import { Video, ResizeMode, AVPlaybackSource } from 'expo-av';
import * as FileSystem from 'expo-file-system';
import * as MediaLibrary from 'expo-media-library';
import * as IntentLauncher from 'expo-intent-launcher';
import { IntentModule } from 'expo-intent-launcher';

interface VideoPlayerProps {
  uri: string;
  onPlaybackError?: (error: Error) => void;
}

export const BulletproofVideoPlayer: React.FC<VideoPlayerProps> = ({ 
  uri, 
  onPlaybackError 
}) => {
  const [playbackSource, setPlaybackSource] = useState<AVPlaybackSource | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [attempt, setAttempt] = useState(1);
  const videoRef = useRef<Video>(null);
  const maxAttempts = 5;

  useEffect(() => {
    loadVideoWithFallback(uri);
  }, [uri]);

  // LAYER 1: Try direct content:// URI
  const tryDirectPlay = async (sourceUri: string): Promise<boolean> => {
    try {
      console.log('🎯 Attempt 1: Direct content:// URI');
      
      // Ensure proper URI format
      const normalizedUri = sourceUri.startsWith('file://') 
        ? sourceUri 
        : sourceUri.startsWith('content://')
        ? sourceUri
        : `file://${sourceUri}`;

      const source: AVPlaybackSource = { uri: normalizedUri };
      setPlaybackSource(source);
      
      // Wait 3 seconds to see if it plays
      await new Promise(resolve => setTimeout(resolve, 3000));
      
      const status = await videoRef.current?.getStatusAsync();
      if (status?.isLoaded && !status.didJustFinish) {
        console.log('✅ Direct play successful');
        return true;
      }
      
      return false;
    } catch (error) {
      console.log('❌ Direct play failed:', error);
      return false;
    }
  };

  // LAYER 2: Copy to cache and play
  const tryCacheCopy = async (sourceUri: string): Promise<boolean> => {
    try {
      console.log('🎯 Attempt 2: Copy to cache directory');
      
      const cacheUri = `${FileSystem.cacheDirectory}status_${Date.now()}.mp4`;
      
      // Copy file to cache (works with content:// URIs)
      await FileSystem.copyAsync({
        from: sourceUri,
        to: cacheUri,
      });

      const source: AVPlaybackSource = { uri: cacheUri };
      setPlaybackSource(source);
      
      await new Promise(resolve => setTimeout(resolve, 2000));
      const status = await videoRef.current?.getStatusAsync();
      
      if (status?.isLoaded) {
        console.log('✅ Cache copy successful');
        return true;
      }
      
      return false;
    } catch (error) {
      console.log('❌ Cache copy failed:', error);
      return false;
    }
  };

  // LAYER 3: Stream via local file server (advanced)
  const tryLocalStream = async (sourceUri: string): Promise<boolean> => {
    try {
      console.log('🎯 Attempt 3: Local file stream');
      
      // Read file as base64 and create data URI
      const base64 = await FileSystem.readAsStringAsync(sourceUri, {
        encoding: FileSystem.EncodingType.Base64,
      });
      
      const dataUri = `data:video/mp4;base64,${base64}`;
      const source: AVPlaybackSource = { uri: dataUri };
      setPlaybackSource(source);
      
      await new Promise(resolve => setTimeout(resolve, 2000));
      const status = await videoRef.current?.getStatusAsync();
      
      if (status?.isLoaded) {
        console.log('✅ Local stream successful');
        return true;
      }
      
      return false;
    } catch (error) {
      console.log('❌ Local stream failed:', error);
      return false;
    }
  };

  // LAYER 4: Open with native video player
  const tryNativePlayer = async (sourceUri: string): Promise<boolean> => {
    try {
      console.log('🎯 Attempt 4: Native player intent');
      
      ToastAndroid.show('Opening in video player...', ToastAndroid.SHORT);
      
      await IntentLauncher.startActivityAsync(IntentLauncher.ActivityActionView, {
        data: sourceUri,
        type: 'video/*',
        flags: 1, // FLAG_GRANT_READ_URI_PERMISSION
      });
      
      console.log('✅ Native player launched');
      return true; // Consider this success (user watches externally)
    } catch (error) {
      console.log('❌ Native player failed:', error);
      return false;
    }
  };

  // LAYER 5: Download to permanent storage and retry
  const tryPermanentDownload = async (sourceUri: string): Promise<boolean> => {
    try {
      console.log('🎯 Attempt 5: Permanent download');
      
      const downloadDir = `${FileSystem.documentDirectory}downloads/`;
      await FileSystem.makeDirectoryAsync(downloadDir, { intermediates: true });
      
      const permanentUri = `${downloadDir}status_${Date.now()}.mp4`;
      
      await FileSystem.copyAsync({
        from: sourceUri,
        to: permanentUri,
      });
      
      const source: AVPlaybackSource = { uri: permanentUri };
      setPlaybackSource(source);
      
      await new Promise(resolve => setTimeout(resolve, 2000));
      const status = await videoRef.current?.getStatusAsync();
      
      if (status?.isLoaded) {
        console.log('✅ Permanent download successful');
        ToastAndroid.show('Video downloaded and ready', ToastAndroid.LONG);
        return true;
      }
      
      return false;
    } catch (error) {
      console.log('❌ Permanent download failed:', error);
      return false;
    }
  };

  // MAIN FALLBACK CHAIN
  const loadVideoWithFallback = async (sourceUri: string) => {
    setIsLoading(true);
    setAttempt(1);

    console.log('🎬 Starting bulletproof video load:', sourceUri);

    // Try each layer until one works
    const strategies = [
      tryDirectPlay,
      tryCacheCopy,
      tryLocalStream,
      tryNativePlayer,
      tryPermanentDownload,
    ];

    for (let i = 0; i < strategies.length; i++) {
      const strategy = strategies[i];
      const success = await strategy(sourceUri);
      
      if (success) {
        setIsLoading(false);
        return;
      }
      
      setAttempt(i + 2);
    }

    // ALL STRATEGIES FAILED
    setIsLoading(false);
    ToastAndroid.show('Unable to play this video', ToastAndroid.LONG);
    onPlaybackError?.(new Error('All playback strategies failed'));
  };

  // Handle playback errors
  const handleError = async (error: any) => {
    console.error('Playback error:', error);
    
    if (attempt < maxAttempts) {
      console.log(`⚠️ Retrying... (Attempt ${attempt + 1}/${maxAttempts})`);
      await loadVideoWithFallback(uri);
    }
  };

  if (isLoading) {
    return (
      <View style={{ 
        flex: 1, 
        backgroundColor: '#000', 
        justifyContent: 'center', 
        alignItems: 'center' 
      }}>
        <ActivityIndicator size="large" color="#4CAF50" />
        <View style={{ marginTop: 10 }}>
          <Text style={{ color: '#fff', fontSize: 12 }}>
            Loading video (Attempt {attempt}/5)...
          </Text>
        </View>
      </View>
    );
  }

  return (
    <View style={{ flex: 1, backgroundColor: '#000' }}>
      {playbackSource && (
        <Video
          ref={videoRef}
          source={playbackSource}
          style={{ flex: 1 }}
          useNativeControls
          resizeMode={ResizeMode.COVER}
          shouldPlay={true}
          onError={handleError}
          onPlaybackStatusUpdate={(status) => {
            if (status.isLoaded && !status.isPlaying) {
              console.log('⚠️ Video loaded but not playing');
            }
          }}
          videoNativeProps={{
            androidHardwareAccelerationDisabled: false,
            androidDisableFullscreenPlayback: false,
          }}
        />
      )}
    </View>
  );
};
```

## 🔍 **STEP 5: CREATE WHATSAPP STATUS DISCOVERY UTILITY**

Create `src/utils/whatsappStatusFinder.ts`:

```typescript
import * as MediaLibrary from 'expo-media-library';
import * as FileSystem from 'expo-file-system';
import { Platform } from 'react-native';

export interface StatusItem {
  id: string;
  uri: string;
  type: 'image' | 'video';
  timestamp: number;
  filename: string;
  duration?: number;
}

export const WhatsAppStatusFinder = {
  // Get all WhatsApp statuses with proper URIs
  getAllStatuses: async (): Promise<StatusItem[]> => {
    try {
      // Request permissions
      const { granted } = await MediaLibrary.requestPermissionsAsync();
      if (!granted) {
        throw new Error('Storage permission denied');
      }

      if (Platform.OS === 'android') {
        return await getAndroidStatuses();
      } else {
        return await getIOSStatuses();
      }
    } catch (error) {
      console.error('Error fetching statuses:', error);
      return [];
    }
  },
};

// ANDROID-SPECIFIC STATUS DISCOVERY
const getAndroidStatuses = async (): Promise<StatusItem[]> => {
  const statuses: StatusItem[] = [];

  try {
    // Method 1: Use MediaLibrary to find WhatsApp albums
    const albums = await MediaLibrary.getAlbumsAsync({
      includeSmartAlbums: false,
    });

    // Find WhatsApp-related albums
    const whatsappAlbums = albums.filter(album => 
      album.title.toLowerCase().includes('whatsapp') ||
      album.title.toLowerCase().includes('status')
    );

    for (const album of whatsappAlbums) {
      const assets = await MediaLibrary.getAssetsAsync({
        album,
        mediaType: ['video', 'photo'],
        first: 100,
        sortBy: MediaLibrary.SortBy.creationTime,
      });

      assets.assets.forEach(asset => {
        statuses.push({
          id: asset.id,
          uri: asset.uri, // This is a proper content:// URI
          type: asset.mediaType === 'video' ? 'video' : 'image',
          timestamp: asset.creationTime,
          filename: asset.filename || `status_${asset.id}`,
          duration: asset.duration,
        });
      });
    }

    // Method 2: Direct file system access (fallback)
    if (statuses.length === 0) {
      console.log('No albums found, trying direct path...');
      const statusPath = '/storage/emulated/0/WhatsApp/Media/.Statuses';
      
      try {
        const files = await FileSystem.readDirectoryAsync(statusPath);
        
        for (const file of files) {
          const filePath = `${statusPath}/${file}`;
          const fileInfo = await FileSystem.getInfoAsync(filePath);
          
          if (fileInfo.exists) {
            const isVideo = file.endsWith('.mp4') || file.endsWith('.3gp');
            statuses.push({
              id: file,
              uri: `content://${filePath}`, // Convert to content://
              type: isVideo ? 'video' : 'image',
              timestamp: fileInfo.modificationTime || Date.now(),
              filename: file,
              duration: isVideo ? 30 : undefined, // Estimate
            });
          }
        }
      } catch (err) {
        console.log('Direct path access failed:', err);
      }
    }

  } catch (error) {
    console.error('Error in getAndroidStatuses:', error);
  }

  return statuses;
};

// iOS FALLBACK
const getIOSStatuses = async (): Promise<StatusItem[]> => {
  // iOS doesn't have WhatsApp status folder
  // You'd need to implement a different strategy
  return [];
};

// Convert file:// to content:// for Android 11+
export const convertToContentUri = async (filePath: string): Promise<string> => {
  if (Platform.OS !== 'android') return filePath;
  
  if (filePath.startsWith('content://')) return filePath;
  
  // For Android 11+, we MUST use content:// URIs
  try {
    // Try to get a content URI via MediaLibrary
    const asset = await MediaLibrary.createAssetAsync(filePath);
    return asset.uri;
  } catch (error) {
    console.log('Failed to create content URI:', error);
    return filePath; // Fallback (may not work on Android 11+)
  }
};
```

const emergencyDownload = async (statusUri: string) => {
  const { granted } = await MediaLibrary.requestPermissionsAsync();
  if (!granted) return;

  try {
    // Download to user's gallery
    const asset = await MediaLibrary.createAssetAsync(statusUri);
    const album = await MediaLibrary.getDefaultAlbumAsync();
    
    if (album) {
      await MediaLibrary.addAssetsToAlbumAsync([asset], album, false);
      ToastAndroid.show('Video saved to gallery!', ToastAndroid.LONG);
    }
  } catch (error) {
    console.error('Emergency download failed:', error);
    ToastAndroid.show('Could not save video', ToastAndroid.LONG);
  }
};
