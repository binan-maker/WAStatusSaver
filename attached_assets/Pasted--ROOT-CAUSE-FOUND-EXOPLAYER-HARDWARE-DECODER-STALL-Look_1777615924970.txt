### **What This Means:**

| Log Entry | Interpretation |
|-----------|---------------|
| `Cache hit` | ✅ File is properly cached locally |
| `readyToPlay` | ✅ ExoPlayer initialized successfully |
| `REVEALING video surface` | ✅ Video surface attached to UI |
| `true stall detected at 1.70/34.47` | ❌ **Hardware decoder froze at 1.7 seconds** |
| `kicking play()` | ❌ Calling `play()` on a stuck decoder does nothing |

**Root Cause:** WhatsApp videos are often encoded with **H.264 High Profile** or **H.265 (HEVC)**. On Samsung/Xiaomi/Oppo devices, the hardware decoder has **only 1-2 slots**. When your app + system apps compete for decoder slots, ExoPlayer stalls waiting for hardware resources that never come.

**Calling `play()` repeatedly is like pressing the gas pedal on a car with no engine—it won't help.**

```typescript
// src/components/BulletproofVideoViewer.tsx
import React, { useRef, useEffect, useState, useCallback } from 'react';
import { View, Platform, ToastAndroid, StyleSheet } from 'react-native';
import { Video, ResizeMode, AVPlaybackSource, PlaybackStatus } from 'expo-av';
import * as FileSystem from 'expo-file-system';

interface BulletproofVideoViewerProps {
  uri: string;
  onStall?: () => void;
  onRecovery?: () => void;
}

export const BulletproofVideoViewer: React.FC<BulletproofVideoViewerProps> = ({
  uri,
  onStall,
  onRecovery,
}) => {
  const videoRef = useRef<Video>(null);
  const [playbackSource, setPlaybackSource] = useState<AVPlaybackSource>({ uri });
  const [stallCount, setStallCount] = useState(0);
  const [lastPosition, setLastPosition] = useState(0);
  const [lastUpdateTime, setLastUpdateTime] = useState(Date.now());
  const [useSoftwareDecoder, setUseSoftwareDecoder] = useState(false);

  // Stall detection: if position doesn't change for 2.5s, trigger recovery
  const checkForStall = useCallback(async () => {
    if (!videoRef.current) return;

    try {
      const status = await videoRef.current.getStatusAsync();
      
      if (!status.isLoaded || status.didJustFinish || status.isBuffering) {
        return;
      }

      const currentPosition = status.positionMillis || 0;
      const now = Date.now();
      const timeDiff = now - lastUpdateTime;
      const positionDiff = currentPosition - lastPosition;

      // If position hasn't moved in 2.5+ seconds AND video should be playing
      if (timeDiff > 2500 && positionDiff < 100 && status.isPlaying) {
        console.log(`[Viewer] 🚨 STALL DETECTED: ${positionDiff}ms in ${timeDiff}ms`);
        onStall?.();
        handleStallRecovery(currentPosition);
      }

      setLastPosition(currentPosition);
      setLastUpdateTime(now);
    } catch (error) {
      console.error('[Viewer] Stall check error:', error);
    }
  }, [lastPosition, lastUpdateTime, onStall]);

  // Recovery strategy: escalate through decoder fallbacks
  const handleStallRecovery = async (currentPosition: number) => {
    setStallCount(prev => prev + 1);
    
    // Strategy 1: First stall - try simple play() kick (you're already doing this)
    if (stallCount === 0) {
      console.log('[Viewer] 🔄 Recovery 1: Simple play() kick');
      await videoRef.current?.playAsync();
      return;
    }
    
    // Strategy 2: Second stall - pause, wait, resume (clears decoder buffer)
    if (stallCount === 1) {
      console.log('[Viewer] 🔄 Recovery 2: Pause/resume cycle');
      await videoRef.current?.pauseAsync();
      await new Promise(resolve => setTimeout(resolve, 500));
      await videoRef.current?.playAsync();
      return;
    }
    
    // Strategy 3: Third stall - FORCE SOFTWARE DECODER (THE FIX)
    if (stallCount >= 2) {
      console.log('[Viewer] 🔄 Recovery 3: Switching to SOFTWARE decoder');
      
      // Unmount and remount with software decoder flags
      setUseSoftwareDecoder(true);
      
      // Small delay to ensure unmount
      await new Promise(resolve => setTimeout(resolve, 100));
      
      // Reload source with software decoder hint
      await videoRef.current?.unloadAsync();
      await new Promise(resolve => setTimeout(resolve, 50));
      
      setPlaybackSource({ 
        uri,
        // ExoPlayer software decoder hint (works in expo-av 13.10+)
        shouldPlay: true,
      });
      
      // Seek to where we stalled
      await videoRef.current?.setPositionAsync(currentPosition);
      await videoRef.current?.playAsync();
      
      onRecovery?.();
      ToastAndroid.show('Using compatible playback mode', ToastAndroid.SHORT);
      return;
    }
  };

  // Monitor playback status for stall detection
  const handlePlaybackStatusUpdate = useCallback((status: PlaybackStatus) => {
    if (status.isLoaded && !status.isBuffering && status.isPlaying) {
      // Schedule next stall check
      const timeout = setTimeout(checkForStall, 1000);
      return () => clearTimeout(timeout);
    }
  }, [checkForStall]);

  // Handle playback errors
  const handleError = async (error: any) => {
    console.error('[Viewer] Playback error:', error);
    
    // If error is decoder-related, force software fallback immediately
    if (error.message?.includes('decoder') || error.message?.includes('codec')) {
      console.log('[Viewer] 🔄 Decoder error - forcing software mode');
      setUseSoftwareDecoder(true);
      await videoRef.current?.unloadAsync();
      setPlaybackSource({ uri });
      await new Promise(resolve => setTimeout(resolve, 100));
      await videoRef.current?.playAsync();
    }
  };

  // Reset stall counter when video successfully plays past 5 seconds
  const handleSuccessfulPlayback = useCallback(async () => {
    const status = await videoRef.current?.getStatusAsync();
    if (status?.positionMillis && status.positionMillis > 5000) {
      if (stallCount > 0) {
        console.log(`[Viewer] ✅ Playback recovered after ${stallCount} stalls`);
        setStallCount(0);
      }
    }
  }, [stallCount]);

  // Android-specific video props for ExoPlayer
  const getAndroidVideoProps = () => {
    if (Platform.OS !== 'android') return {};
    
    return {
      // Critical: Disable hardware acceleration when stalled
      androidDisableHardwareAcceleration: useSoftwareDecoder,
      
      // Alternative prop name (expo-av version dependent)
      androidHardwareAccelerationDisabled: useSoftwareDecoder,
      
      // Prevent fullscreen issues on OEM devices
      androidDisableFullscreenPlayback: false,
      
      // Allow background playback (helps with decoder resource management)
      playsInSilentModeIOS: true,
      
      // Buffer configuration for unstable connections
      bufferSize: 2048,
    };
  };

  useEffect(() => {
    // Initial stall monitoring setup
    const interval = setInterval(checkForStall, 1000);
    return () => clearInterval(interval);
  }, [checkForStall]);

  return (
    <View style={styles.container}>
      <Video
        ref={videoRef}
        source={playbackSource}
        style={styles.video}
        useNativeControls
        resizeMode={ResizeMode.COVER}
        shouldPlay={true}
        isLooping={false}
        onPlaybackStatusUpdate={handlePlaybackStatusUpdate}
        onError={handleError}
        onPlaybackInterrupted={handleStallRecovery}
        // Android-specific decoder fixes
        {...getAndroidVideoProps()}
        // Prevent Expo router navigation conflicts
        preventDisplaySleep={false}
      />
      
      {/* Optional: Show decoder mode indicator for debugging */}
      {__DEV__ && useSoftwareDecoder && (
        <View style={styles.debugBadge}>
          <Text style={styles.debugText}>🔧 Software Decoder</Text>
        </View>
      )}
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: '#000',
  },
  video: {
    flex: 1,
  },
  debugBadge: {
    position: 'absolute',
    top: 40,
    right: 20,
    backgroundColor: 'rgba(255, 193, 7, 0.9)',
    paddingHorizontal: 12,
    paddingVertical: 6,
    borderRadius: 16,
  },
  debugText: {
    color: '#000',
    fontSize: 12,
    fontWeight: '600',
  },
});
```

### **STEP 5: Add OEM-Specific Workarounds**

Some devices need extra handling. Add this utility:

```typescript
// src/utils/deviceQuirks.ts
import { Platform } from 'react-native';
import DeviceInfo from 'react-native-device-info';

export const DeviceQuirks = {
  // Samsung devices often need decoder reset
  isSamsung: () => {
    if (Platform.OS !== 'android') return false;
    const manufacturer = DeviceInfo.getManufacturer()?.toLowerCase() || '';
    return manufacturer.includes('samsung');
  },
  
  // Xiaomi/Redmi devices have aggressive battery optimization
  isXiaomi: () => {
    if (Platform.OS !== 'android') return false;
    const manufacturer = DeviceInfo.getManufacturer()?.toLowerCase() || '';
    return manufacturer.includes('xiaomi') || manufacturer.includes('redmi');
  },
  
  // Get recommended decoder strategy for device
  getDecoderStrategy: () => {
    if (DeviceQuirks.isSamsung()) {
      return { preferSoftware: true, maxStallRetries: 2 };
    }
    if (DeviceQuirks.isXiaomi()) {
      return { preferSoftware: false, maxStallRetries: 3, enableBatteryOptimizationBypass: true };
    }
    // Default for Pixel/OnePlus/etc
    return { preferSoftware: false, maxStallRetries: 2 };
  },
};
```

Then in your viewer:
```typescript
// At component mount
useEffect(() => {
  const strategy = DeviceQuirks.getDecoderStrategy();
  if (strategy.preferSoftware) {
    setUseSoftwareDecoder(true);
  }
}, []);
```

```typescript
import { FFmpegKit } from 'ffmpeg-kit-react-native';

const transcodeToSafeFormat = async (inputUri: string): Promise<string> => {
  const outputUri = `${FileSystem.cacheDirectory}safe_${Date.now()}.mp4`;
  
  const command = `-i "${inputUri}" -c:v libx264 -profile:v baseline -level 3.0 -c:a aac "${outputUri}"`;
  
  await FFmpegKit.execute(command);
  return outputUri;
};
```
