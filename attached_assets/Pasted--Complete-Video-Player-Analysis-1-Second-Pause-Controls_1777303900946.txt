## 🎯 Critical Issue #1: Player Instance Not Persisted

**Your Current Code Pattern (`viewer.tsx` ~line 640):**
```tsx
const ViewerItem = ({ item, isActive, isNearActive }) => {
  // ❌ PROBLEM: Player created fresh on every render
  const player = useMemo(() => new Player(), []);
  
  useEffect(() => {
    if (isActive && item.type === 'video') {
      player.replace({ uri: item.uri }); // Called every time isActive flips
    }
  }, [isActive, item.uri, player]);
  
  return <VideoView player={player} ... />;
};
```

**Why Videos Pause After 1 Second:**
1. User swipes → `isActive` flips from `true` → `false` for previous item
2. `useEffect` cleanup runs → `player.pauseAsync()` or `player.unloadAsync()` called
3. Next swipe → `isActive` flips back → new `replace()` call
4. Android 11+ hardware decoder needs 200-400ms to re-allocate → video stalls
5. If `PlaybackState` isn't `ReadyToPlay`, `nativeControls` never appear

## 🎯 Critical Issue #2: Controls Require Active Playback

**expo-video Behavior:**
```tsx
<VideoView
  player={player}
  nativeControls={true} // ✅ Correct
/>
```

**The Reality:**
- `nativeControls` only render when `player.playbackState === PlaybackState.Playing`
- If video is `Paused`, `Buffering`, or `Idle` → controls stay hidden
- Your code doesn't explicitly call `player.playAsync()` after `replace()`

## ✅ Production Fix: Persistent Player Pool + Playback Management

### Step 1: Create a Player Pool Hook (Prevents Recreation)

```typescript
// hooks/useVideoPlayerPool.ts
import { Player, PlaybackState } from 'expo-video';
import { useRef, useEffect, useCallback } from 'react';

const MAX_CACHED_PLAYERS = 3; // Android 11+ decoder limit

export function useVideoPlayerPool() {
  const playerPool = useRef<Map<string, Player>>(new Map());
  const activePlayerRef = useRef<Player | null>(null);

  const getPlayer = useCallback((itemId: string): Player => {
    // Return existing player if cached
    if (playerPool.current.has(itemId)) {
      return playerPool.current.get(itemId)!;
    }
    
    // Create new player if pool has space
    if (playerPool.current.size < MAX_CACHED_PLAYERS) {
      const player = new Player();
      playerPool.current.set(itemId, player);
      return player;
    }
    
    // Reuse least-recently-used player if pool is full
    const lruId = playerPool.current.keys().next().value;
    const reused = playerPool.current.get(lruId)!;
    playerPool.current.delete(lruId);
    playerPool.current.set(itemId, reused);
    return reused;
  }, []);

  const releasePlayer = useCallback((itemId: string) => {
    const player = playerPool.current.get(itemId);
    if (player) {
      // Pause but don't unload → keeps decoder warm for quick reuse
      player.pauseAsync().catch(() => {});
      // Only fully unload if pool is at capacity and we need space
      if (playerPool.current.size >= MAX_CACHED_PLAYERS) {
        player.unloadAsync().catch(() => {});
        playerPool.current.delete(itemId);
      }
    }
  }, []);

  const cleanup = useCallback(() => {
    playerPool.current.forEach(player => {
      player.unloadAsync().catch(() => {});
    });
    playerPool.current.clear();
  }, []);

  useEffect(() => {
    return () => cleanup();
  }, [cleanup]);

  return { getPlayer, releasePlayer, activePlayerRef };
}
```

### Step 2: Fix `ViewerItem` with Proper Playback Lifecycle

```tsx
// app/viewer.tsx — Replace ViewerItem component

import { Player, VideoView, PlaybackState, InterruptionModeAndroid } from 'expo-video';
import { useVideoPlayerPool } from '@/hooks/useVideoPlayerPool';

const ViewerItem = ({ 
  item, 
  isActive, 
  isNearActive, 
  prepareStatusForViewing,
  onToggleControls 
}) => {
  const { getPlayer, releasePlayer } = useVideoPlayerPool();
  const [playbackState, setPlaybackState] = useState<PlaybackState>(PlaybackState.Idle);
  const [hasError, setHasError] = useState(false);
  const playerRef = useRef<Player | null>(null);
  const mountedRef = useRef(true);

  // 🔧 CRITICAL: Get persistent player instance
  useEffect(() => {
    if (item.type !== 'video') return;
    
    playerRef.current = getPlayer(item.id);
    
    // Subscribe to playback state for controls visibility
    const unsubscribe = playerRef.current.addListener(({ playbackState: newState }) => {
      if (!mountedRef.current) return;
      setPlaybackState(newState);
      
      // 🔧 CRITICAL: Show native controls only when ready + playing
      if (newState === PlaybackState.ReadyToPlay || newState === PlaybackState.Playing) {
        // Optional: auto-show controls for 3 seconds on load
        setTimeout(() => {
          if (mountedRef.current) onToggleControls?.(false);
        }, 3000);
      }
    });

    return () => {
      unsubscribe?.();
      if (!isNearActive) {
        releasePlayer(item.id);
      }
    };
  }, [item.id, item.type, getPlayer, releasePlayer, isNearActive, onToggleControls]);

  // 🔧 CRITICAL: Load video ONLY when active + player ready
  useEffect(() => {
    if (item.type !== 'video' || !isActive || !playerRef.current) return;
    
    let cancelled = false;
    
    const loadVideo = async () => {
      try {
        const uri = await prepareStatusForViewing(item, { forPlayback: true });
        if (cancelled) return;
        
        // Only replace if URI changed or player is idle
        if (playerRef.current?.source?.uri !== uri) {
          await playerRef.current.replace({ uri });
        }
        
        // 🔧 CRITICAL: Explicitly play after replace (required for nativeControls)
        await playerRef.current.playAsync();
        
        // Android-specific: ensure audio focus doesn't duck other apps unexpectedly
        if (Platform.OS === 'android') {
          playerRef.current.setInterruptionModeAsync?.(InterruptionModeAndroid.DuckOthers);
        }
      } catch (e) {
        console.error('[Viewer] Video load failed:', e);
        if (!cancelled) setHasError(true);
      }
    };
    
    loadVideo();
    
    return () => {
      cancelled = true;
      // 🔧 CRITICAL: Pause but don't unload on deactivation (keeps decoder warm)
      if (!isActive && playerRef.current) {
        playerRef.current.pauseAsync().catch(() => {});
      }
    };
  }, [item, isActive, prepareStatusForViewing]);

  if (item.type !== 'video') return null;

  return (
    <View style={{ width: '100%', height: '100%', backgroundColor: '#000' }}>
      <VideoView
        player={playerRef.current!}
        style={{ width: '100%', height: '100%' }}
        contentFit="contain"
        allowsFullscreen
        // 🔧 CRITICAL: nativeControls appear ONLY when playbackState is Playing/Ready
        nativeControls={playbackState === PlaybackState.Playing || playbackState === PlaybackState.ReadyToPlay}
        // Android SAF compatibility
        {...(Platform.OS === 'android' && {
          // expo-video handles ContentResolver internally for content:// URIs
          // No extra props needed beyond proper permission granting in prepareStatusForViewing
        })}
      />
      
      {/* Loading overlay while buffering */}
      {(playbackState === PlaybackState.Loading || playbackState === PlaybackState.Buffering) && (
        <View style={StyleSheet.absoluteFill} pointerEvents="none">
          <ActivityIndicator size="large" color="#fff" />
        </View>
      )}
      
      {/* Error retry UI */}
      {hasError && (
        <TouchableOpacity 
          style={[StyleSheet.absoluteFill, { backgroundColor: 'rgba(0,0,0,0.7)', alignItems: 'center', justifyContent: 'center' }]}
          onPress={() => {
            setHasError(false);
            // Re-trigger load effect
            if (playerRef.current) {
              playerRef.current.replace({ uri: item.uri }).catch(() => {});
              playerRef.current.playAsync().catch(() => {});
            }
          }}
        >
          <Ionicons name="refresh-outline" size={32} color="#fff" />
          <Text style={{ color: '#fff', marginTop: 8 }}>Tap to retry</Text>
        </TouchableOpacity>
      )}
    </View>
  );
};
```

### Step 4: Ensure `prepareStatusForViewing` Returns Playable URI

```typescript
// In MediaContext.tsx or media-store.ts
const prepareStatusForViewing = useCallback(async (
  item: StatusItem,
  opts?: { forShare?: boolean; forPlayback?: boolean },
): Promise<string> => {
  if (!item.uri.startsWith('content://')) return item.uri;

  // Playback: return URI immediately + start background copy for fallback
  if (!opts?.forShare) {
    if (opts?.forPlayback) {
      // Start background copy silently (fallback if direct playback fails)
      enqueueCopy(async () => {
        const ext = item.name.split('.').pop() || 'mp4';
        const safeId = item.id.replace(/[^a-z0-9]/gi, '_');
        const cacheUri = `${FileSystem.cacheDirectory}playback_${safeId}.${ext}`;
        const info = await FileSystem.getInfoAsync(cacheUri).catch(() => ({ exists: false }));
        if (!info.exists) {
          await FileSystem.copyAsync({ from: item.uri, to: cacheUri }).catch(() => {});
        }
      });
      return item.uri; // expo-video handles content:// with granted SAF permission
    }
    return item.uri;
  }

  // Share path: copy to file:// (share sheet needs real file)
  const ext = item.name.split('.').pop() || 'jpg';
  const safeId = item.id.replace(/[^a-z0-9]/gi, '_');
  const tempUri = `${FileSystem.cacheDirectory}share_${safeId}.${ext}`;
  
  try {
    const info = await FileSystem.getInfoAsync(tempUri);
    if (info.exists && (info as any).size > 0) return tempUri;
  } catch {}

  return enqueueCopy(async () => {
    await FileSystem.copyAsync({ from: item.uri, to: tempUri });
    return tempUri;
  });
}, []);
```
